const geminiModel='https://generativelanguage.googleapis.com/v1beta/models/gemini-3.5-flash-lite';
async function activeVoice(env){
 const active=await env.BUCKET.get('private/voice-provider');
 if(active)return active.text();
 return env.GROQ_API_KEY||await env.BUCKET.head(keyObject)?'Groq':'';
}
async function geminiOrder(chunks,key,reply){
 const bytes=new Uint8Array(await new Blob(chunks).arrayBuffer());
 let binary='';
 for(let i=0;i<bytes.length;i+=8192)binary+=String.fromCharCode(...bytes.subarray(i,i+8192));
 const schema={type:'OBJECT',properties:{text:{type:'STRING'},items:{type:'ARRAY',items:{type:'OBJECT',properties:{
 ta:{type:'STRING'},en:{type:'STRING'},amount:{type:'NUMBER',nullable:true},unit:{type:'STRING',enum:['','kg','g','L','ml','packet','piece','box','bottle','jar','dozen']}
 },required:['ta','en','amount','unit']}}},required:['text','items']};
 const body={contents:[{parts:[{text:'Transcribe this Tamil, English or mixed grocery order verbatim in text. Extract each requested product in spoken order with Tamil and English names. Preserve brands and quantities. Use null for an unspoken or unclear quantity, empty unit if unspecified. Do not invent items. For silence or unrelated speech return an empty items array.'},{inlineData:{mimeType:'audio/wav',data:btoa(binary)}}]}],generationConfig:{responseMimeType:'application/json',responseSchema:schema}};
 if(reply){
   body.contents[0].parts[0].text='Transcribe the short Tamil answer or customer name exactly as spoken into text. Use an empty string for silence.';
   body.generationConfig.responseSchema={type:'OBJECT',properties:{text:{type:'STRING'}},required:['text']};
 }
 const r=await fetch(geminiModel+':generateContent',{method:'POST',headers:{'x-goog-api-key':key,'Content-Type':'application/json'},body:JSON.stringify(body)});
 if(!r.ok)return new Response('Gemini request failed.',{status:r.status===429?429:502});
 try{
   const data=await r.json();
   const order=JSON.parse(data.candidates[0].content.parts.filter(p=>!p.thought).map(p=>p.text||'').join(''));
   if(reply && typeof order.text==='string')return Response.json({text:order.text},{headers:{'Cache-Control':'no-store'}});
   if(typeof order.text!=='string'||!Array.isArray(order.items)||order.items.some(i=>!i||typeof i.ta!=='string'||typeof i.en!=='string'||typeof i.unit!=='string'||(i.amount!==null&&!(typeof i.amount==='number'&&i.amount>0))))throw Error();
   return Response.json(order,{headers:{'Cache-Control':'no-store'}});
 }catch{return new Response('Could not read the item list. Please retry.',{status:502})}
}
const keyObject='private/groq-key';
async function encryptionKey(ownerCode){
 const digest=await crypto.subtle.digest('SHA-256',new TextEncoder().encode('Kural Orders Groq secret:'+ownerCode));
 return crypto.subtle.importKey('raw',digest,'AES-GCM',false,['encrypt','decrypt']);
}
const providerKeys={Groq:keyObject,Gemini:'private/gemini-key',Sarvam:'private/sarvam-key'};
async function configureVoice(req,env,provider){
 if(req.method!=='POST')return new Response('Method not allowed',{status:405});
 if(!env.OWNER_CODE||req.headers.get('Authorization')!=='Bearer '+env.OWNER_CODE)return new Response('Incorrect owner access code.',{status:403});
 const key=(await req.text()).trim();
 if(!key||key.length>256||(provider==='Groq'&&!key.startsWith('gsk_')))return new Response('Enter a valid API key.',{status:400});
 if(provider!=='Sarvam'){
 const verify=provider==='Gemini'?await fetch(geminiModel,{headers:{'x-goog-api-key':key}}):await fetch('https://api.groq.com/openai/v1/models/whisper-large-v3',{headers:{Authorization:'Bearer '+key}});
 if(!verify.ok)return new Response('Could not verify model access. Check the key and model permissions.',{status:400});
 }
 const iv=crypto.getRandomValues(new Uint8Array(12));
 const cipher=await crypto.subtle.encrypt({name:'AES-GCM',iv},await encryptionKey(env.OWNER_CODE),new TextEncoder().encode(key));
 await env.BUCKET.put(providerKeys[provider]||keyObject,JSON.stringify({iv:Array.from(iv),data:Array.from(new Uint8Array(cipher))}));

 await env.BUCKET.put('private/voice-provider',provider);
 if(provider==='Sarvam')return new Response('Sarvam selected. Record a short order to test your key.');
 return new Response(provider+' connected and active for voice orders.');
}
async function transcribe(req,env){
 if(req.method!=='POST')return new Response('Method not allowed',{status:405});
 const type=(req.headers.get('Content-Type')||'').split(';')[0];
 const provider=await activeVoice(env);
 const gemini=provider==='Gemini';
 const ext={'audio/wav':'wav','audio/webm':'webm','audio/mp4':'mp4','audio/ogg':'ogg'}[type];
 if(!ext||(gemini&&ext!=='wav'))return new Response('Unsupported recording format.',{status:415});
 const max=4*1024*1024;
 if(Number(req.headers.get('Content-Length'))>max)return new Response('Recording is too large.',{status:413});
 let key=provider==='Groq'?env.GROQ_API_KEY:null;
 if(!key){
   const stored=await env.BUCKET.get(providerKeys[provider]||keyObject);
   if(!stored||!env.OWNER_CODE)return new Response('Voice provider has not been connected.',{status:503});
   const {iv,data}=await stored.json();
   const plain=await crypto.subtle.decrypt({name:'AES-GCM',iv:new Uint8Array(iv)},await encryptionKey(env.OWNER_CODE),new Uint8Array(data));
   key=new TextDecoder().decode(plain);
 }
 const reader=req.body?.getReader(),chunks=[];let size=0;
 if(reader)while(true){
   const {value,done}=await reader.read();if(done)break;
   size+=value.length;if(size>max){await reader.cancel();return new Response('Recording is too large.',{status:413})}
   chunks.push(value);
 }
 if(!size)return new Response('No audio received.',{status:400});
 if(gemini)return geminiOrder(chunks,key,req.headers.get('X-Voice-Purpose')==='reply');
 const payload=new FormData();
 payload.set('file',new Blob(chunks,{type}),'order.'+ext);
 if(provider==='Sarvam'){
   payload.set('model','saaras:v4');
   const english=req.headers.get('X-Speech-Language')==='en';
   payload.set('language_code',english?'en-IN':'ta-IN');
   payload.set('mode',english?'transcribe':'codemix');
   const response=await fetch('https://api.sarvam.ai/speech-to-text',{method:'POST',headers:{'api-subscription-key':key},body:payload});
   if(!response.ok)return new Response('Sarvam transcription failed.',{status:[401,403,429].includes(response.status)?response.status:502});
   const result=await response.json();
   return Response.json({text:typeof result.transcript==='string'?result.transcript.trim():''},{headers:{'Cache-Control':'no-store'}});
 }
 payload.set('model','whisper-large-v3');
 payload.set('language',req.headers.get('X-Speech-Language')==='en'?'en':'ta');
 const upstream=await fetch('https://api.groq.com/openai/v1/audio/transcriptions',{method:'POST',headers:{Authorization:'Bearer '+key},body:payload});
 if(!upstream.ok)return new Response(upstream.status===429?'Groq usage limit reached. Type your order or try again later.':'Groq could not transcribe this recording. Retry or type your order.',{status:upstream.status===429?429:502});
 const {text}=await upstream.json();
 return Response.json({text:typeof text==='string'?text.trim():''},{headers:{'Cache-Control':'no-store'}});
}

 if(url.pathname==='/transcribe')return transcribe(request,env);

 if(request.method!=='POST')return new Response('Method not allowed',{status:405});

 }
